@dataclass
class cell_pairs:

    "Calculate the distance vector between two points ci and cj."
    @property
    def rij(self):
        rij =  tuple(a-b for a, b in zip(self.ci, self.cj)) #returns a tuple representing the distance vector : vector value
        return rij
    "calculate the magnitude of the distance vector"
    
    def dij(self, ci, cj):
        dij = math.sqrt(sum((a - b) ** 2 for a, b in zip(self.ci, self.cj))) #returns the magnitude of the distance vector : scalar value
        return dij
    "calculate the unit vector along the distance vector"
    @property
    def nij(self, dij = None , rij= None):  
        if dij is None:
            dij = self.dij  
        if rij is None:
            rij = self.rij
        nij = tuple (a/dij for a in rij if dij !=0) #unit vector along the distance vector
        return nij

@dataclass
class bookofpositions:
    "Class to store and update the positions of cells over time."

    num_cells: int = len(initstate.cell_centers)
    timesteps: int = int(simulparams.total_time/simulparams.dt) #total_time is defined for 10 time intervals to run.

    positions = {i: np.full(len(bookofpositions.timesteps), np.nan) for i in range(num_cells)}
    
    # Initialize positions at time t=0

    for i in range(num_cells):
            positions[i][0] = initstate.cell_centers[i]
    
    #update positions starting from time interval t=1 to t=timesteps-1
    for t in range(1, timesteps):
        for i in range(num_cells):
            positions[i][t] = new_position.new_pos(i)
